import math
import random
import torch
import collections
from torch.utils.data import Dataset, DataLoader

# ==========================================
# 1. 数据预处理与词表构建类
# ==========================================
class PTBDataProcess:
    """
    处理原始文本，包括分词、构建词表、下采样
    """
    def __init__(self, sentences, min_freq=1):
        # sentences: 原始句子列表，例如 [["i", "love", "coding"], ["he", "is", ...]]
        self.sentences = sentences
        # min_freq: 最小词频，低于这个频率的词会被丢弃
        self.min_freq = min_freq
        
        # 1. 统计词频
        # counter 存储每个词出现的次数
        self.counter = self._count_corpus()
        
        # 2. 构建词表 (Vocab)
        # idx_to_token: 索引到词的映射 list
        # token_to_idx: 词到索引的映射 dict
        self.idx_to_token = []
        self.token_to_idx = {}
        self._build_vocab()
        
        # 3. 转换数据
        # 将原始句子中的单词转换为索引 ID，得到 corpus (List[List[int]])
        self.corpus = self._convert_tokens_to_ids()
        
        # 4. 下采样 (Subsampling)
        # 丢弃过高频率的词（如 "the", "a"），优化训练效果
        self.subsampled_corpus = self._subsample()

    def _count_corpus(self):
        """统计语料库中所有词的频率"""
        # collections.Counter 是一个高效的计数器
        # flattened_tokens 把所有句子的词展平到一个列表中
        tokens = [tk for line in self.sentences for tk in line]
        return collections.Counter(tokens)

    def _build_vocab(self):
        """构建词表映射"""
        # 未知词标记
        self.idx_to_token = ['<unk>']
        self.token_to_idx = {'<unk>': 0}
        0r4
        # 遍历统计结果，只保留频率大于 min_freq 的词
        # sorted 按频率从高到低排序，保证高频词 ID 较小（可选）
        for token, freq in sorted(self.counter.items(), key=lambda x: x[1], reverse=True):
            if freq >= self.min_freq:
                self.idx_to_token.append(token)
                # 当前词的 ID 就是列表的长度
                self.token_to_idx[token] = len(self.idx_to_token) - 1

    def _convert_tokens_to_ids(self):
        """将文本转换为索引列表"""
        corpus = []
        for line in self.sentences:
            # 对于句子中的每个词，如果不在词表中，就转为 <unk> (ID=0)
            line_ids = [self.token_to_idx.get(tk, 0) for tk in line]
            corpus.append(line_ids)
        return corpus

    def _subsample(self):
        """
        下采样：高频词有更高概率被丢弃。
        公式：P(drop) = 1 - sqrt(threshold / freq)
        """
        subsampled = []
        # total_tokens: 总词数，用于计算频率
        total_tokens = sum(len(line) for line in self.corpus)
        # threshold: 阈值，通常取 1e-4
        threshold = 1e-4
        
        for line in self.corpus:
            sub_line = []
            for token_id in line:
                # 获取该词的词频
                count = self.counter[self.idx_to_token[token_id]]
                # 计算频率 fraction
                freq = count / total_tokens
                # 计算保留概率 keep_prob
                # 如果 freq 很大，keep_prob 就会很小
                keep_prob = math.sqrt(threshold / freq)
                
                # random.uniform(0, 1) 生成 0-1 之间的随机数
                # 如果随机数小于保留概率，则保留该词
                if random.uniform(0, 1) < keep_prob:
                    sub_line.append(token_id)
            subsampled.append(sub_line)
        return subsampled

# 2. 核心数据集类 (PyTorch Dataset)
# ==========================================
class Word2VecDataset(Dataset):
    """
    负责生成用于训练的 (Center, Context, Negative) 样本
    """
    def __init__(self, processed_data, window_size=2, num_negatives=5):
        # processed_data: 上面定义的 PTBDataProcess 实例
        self.data = processed_data
        # window_size: 上下文窗口大小 (左右各取多少个)
        self.window_size = window_size
        # num_negatives: 每个正样本对应的负采样数量 K
        self.num_negatives = num_negatives
        
        # 1. 提取所有的中心词和上下文词对
        # self.centers: List[int]
        # self.contexts: List[List[int]] (注意：这里 context 是一个列表)
        self.centers, self.contexts = self._get_centers_and_contexts()
        
        # 2. 生成负样本
        # self.negatives: List[List[int]]
        self.negatives = self._get_negatives()

    def _get_centers_and_contexts(self):
        """提取中心词及其对应的上下文词"""
        centers, contexts = [], []
        # 遍历每一句话
        for line in self.data.subsampled_corpus:
            # 如果句子太短（少于2个词），无法构成上下文，跳过
            if len(line) < 2:
                continue
            
            # 遍历句子中的每一个词作为中心词
            for i in range(len(line)):
                # 随机动态窗口：为了增加随机性，实际窗口大小可以在 [1, window_size] 之间随机
                # 这里为了简单，我们固定使用 window_size，或者你可以 uncomment 下面这行：
                # actual_window = random.randint(1, self.window_size) 
                
                # 定义上下文的索引范围
                start = max(0, i - self.window_size)
                end = min(len(line), i + self.window_size + 1)
                
                # 提取上下文词索引，排除中心词自己
                context_indices = [line[j] for j in range(start, end) if j != i]
                
                # 只有当存在上下文时才加入
                if context_indices:
                    centers.append(line[i])
                    contexts.append(context_indices)
        return centers, contexts

    def _get_negatives(self):
        """
        负采样逻辑
        采样分布：P(w) 正比于 count(w)^0.75
        """
        negatives = []
        # 获取词表中的所有词 ID
        tokens = list(self.data.counter.keys())
        token_ids = [self.data.token_to_idx[t] for t in tokens]
        
        # 计算采样权重：词频的 0.75 次幂
        # 这种加权方式能让低频词有更多机会被采样为负样本
        counts = [self.data.counter[t] for t in tokens]
        sampling_weights = [c**0.75 for c in counts]
        
        # population: 用于采样的候选集合（所有词的ID）
        population = token_ids
        
        # 遍历每一个中心词对应的上下文列表
        # 我们需要为每一个"正上下文"生成 K 个"负样本"
        for ctx_list in self.contexts:
            # 当前中心词需要的负样本总数 = 上下文词数量 * K
            k = len(ctx_list) * self.num_negatives
            
            # 使用 random.choices 进行加权重复采样
            # 这是一个高效的采样方法，weights 参数指定了概率分布
            negs = random.choices(population, weights=sampling_weights, k=k)
            
            # 存入列表
            negatives.append(negs)
        return negatives

    def __getitem__(self, idx):
        """
        PyTorch DataLoader 获取单个样本的接口
        返回: (center_word, context_words, noise_words)
        """
        return (self.centers[idx], self.contexts[idx], self.negatives[idx])

    def __len__(self):
        """返回数据集大小"""
        return len(self.centers)

# 3. Batch 整理函数 (Collate Function)
# ==========================================
def batchify(data):
    """
    自定义的 collate_fn，用于 DataLoader。
    因为每个中心词的上下文长度不同，需要进行 Padding。
    """
    # max_len: 找出当前 Batch 中最长的序列长度（正样本+负样本）
    # data 是一个 list，每个元素是 __getitem__ 返回的 tuple
    max_len = max(len(c) + len(n) for _, c, n in data)
    
    centers, contexts_negatives, masks, labels = [], [], [], []
    
    for center, context, negative in data:
        # 1. 处理中心词：reshape 成 (1, 1) 的形状
        cur_centers = [center]
        centers += cur_centers
        
        # 2. 拼接正样本和负样本
        # 这是一个关键步骤：我们将它们拼在一起，形成一个列表
        cur_contexts_negatives = context + negative
        
        # 3. 生成 Labels
        # 正样本对应的 Label 是 1，负样本对应的 Label 是 0
        cur_labels = [1] * len(context) + [0] * len(negative)
        
        # 4. Padding (填充)
        # 如果当前长度小于 max_len，需要补 0
        while len(cur_contexts_negatives) < max_len:
            cur_contexts_negatives.append(0) # 填充 ID 0 (通常是 <unk> 或 pad)
            cur_labels.append(0)             # 填充部分的 Label 无所谓，因为会被 Mask 掉
        
        # 5. 生成 Mask
        # 有效数据的部分为 1，填充部分为 0
        # 长度为 max_len
        cur_mask = [1] * (len(context) + len(negative)) + [0] * (max_len - len(context) - len(negative))
        
        contexts_negatives.append(cur_contexts_negatives)
        labels.append(cur_labels)
        masks.append(cur_mask)
    
    # 将 list 转换为 Tensor
    # centers shape: (batch_size, 1)
    # contexts_negatives shape: (batch_size, max_len)
    # masks shape: (batch_size, max_len)
    # labels shape: (batch_size, max_len)
    return (torch.tensor(centers).reshape((-1, 1)), 
            torch.tensor(contexts_negatives), 
            torch.tensor(masks), 
            torch.tensor(labels))
